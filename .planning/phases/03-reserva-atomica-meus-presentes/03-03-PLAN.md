```php
<?php
/**
 * public/my-gifts/index.php — "Meus Presentes" page for authenticated guests.
 * Variables: $reservations (array), $e (htmlspecialchars closure from View::render)
 */
?>
<div class="container py-4">

    <div class="d-flex align-items-center justify-content-between mb-4">
        <h1 class="fw-bold mb-0">Meus Presentes</h1>
        <a href="/gifts" class="btn btn-outline-secondary btn-sm">
            <i class="bi bi-arrow-left me-1"></i>Ver lista de presentes
        </a>
    </div>

    <?php if (empty($reservations)): ?>
        <!-- D-13: Empty state -->
        <div class="text-center py-5">
            <i class="bi bi-gift display-1 text-muted"></i>
            <h3 class="mt-3 text-muted">Você ainda não reservou nenhum presente</h3>
            <p class="text-muted mb-4">Escolha um presente na lista e reserve agora!</p>
            <a href="/gifts" class="btn btn-primary">Ver lista de presentes</a>
        </div>
    <?php else: ?>
        <div class="row row-cols-1 row-cols-md-2 row-cols-lg-3 g-4">
            <?php foreach ($reservations as $res): ?>
                <div class="col">
                    <div class="card h-100 shadow-sm">
                        <?php if (!empty($res['image_path'])): ?>
                            <img src="<?= $e($res['image_path']) ?>"
                                 class="card-img-top"
                                 alt="<?= $e($res['gift_name']) ?>"
                                 style="object-fit:cover; height:180px;">
                        <?php else: ?>
                            <div class="bg-light d-flex align-items-center justify-content-center"
                                 style="height:180px;">
                                <i class="bi bi-gift display-4 text-muted"></i>
                            </div>
                        <?php endif; ?>

                        <div class="card-body">
                            <h5 class="card-title fw-semibold"><?= $e($res['gift_name']) ?></h5>
                            <p class="text-muted small mb-2">
                                <i class="bi bi-tag me-1"></i><?= $e($res['category']) ?>
                            </p>
                            <p class="small text-muted mb-2">
                                <i class="bi bi-calendar3 me-1"></i>
                                Reservado em <?= $e(date('d/m/Y', strtotime($res['created_at']))) ?>
                            </p>

                            <?php
                            $statusLabel = match($res['status']) {
                                'reserved'  => ['label' => 'Reservado',  'class' => 'bg-warning text-dark'],
                                'confirmed' => ['label' => 'Confirmado', 'class' => 'bg-success'],
                                'cancelled' => ['label' => 'Cancelado',  'class' => 'bg-secondary'],
                                default     => ['label' => $e($res['status']), 'class' => 'bg-light text-dark'],
                            };
                            ?>
                            <span class="badge <?= $statusLabel['class'] ?>">
                                <?= $statusLabel['label'] ?>
                            </span>
                        </div>

                        <?php if ($res['status'] === 'reserved'): ?>
                            <!-- D-11: Cancel button visible only for status=reserved -->
                            <div class="card-footer bg-transparent border-top-0 pt-0">
                                <form method="POST" action="/my-gifts/cancel"
                                      onsubmit="return confirm('Cancelar esta reserva?');">
                                    <input type="hidden"
                                           name="csrf_token"
                                           value="<?= htmlspecialchars(\App\Middleware\CsrfMiddleware::token(), ENT_QUOTES, 'UTF-8') ?>">
                                    <input type="hidden"
                                           name="reservation_id"
                                           value="<?= $e($res['id']) ?>">
                                    <button type="submit"
                                            class="btn btn-outline-danger btn-sm w-100">
                                        <i class="bi bi-x-circle me-1"></i>Cancelar reserva
                                    </button>
                                </form>
                            </div>
                        <?php endif; ?>
                    </div>
                </div>
            <?php endforeach; ?>
        </div>
    <?php endif; ?>

</div>
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| Browser → GET /my-gifts | Server-rendered page; $reservations is fetched by user_id from session — no user-controlled filter |
| Cancel form → POST /my-gifts/cancel | reservation_id from hidden input is user-controlled |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-12 | Tampering | CSRF on cancel form | mitigate | `CsrfMiddleware::token()` rendered in hidden input; Router validates on POST |
| T-03-13 | Tampering | IDOR via reservation_id in cancel form (user changes hidden value in DevTools) | mitigate | ReservationModel::cancel() always queries `WHERE id = :id AND user_id = :uid` — server enforces ownership |
| T-03-14 | Information Disclosure | reservation data in HTML (gift names, status) | accept | This is the user's own data displayed on their authenticated page; no other user's data is in scope |
| T-03-15 | Elevation of Privilege | Unauthenticated access to /my-gifts | mitigate | AuthMiddleware::requireGuest() in ReservationController::myGifts() (Plan 01); redirects to /gifts |
</threat_model>
